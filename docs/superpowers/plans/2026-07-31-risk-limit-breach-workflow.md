```python
"""Derivation of a monitoring envelope from the latest completed risk run.

Shared by the run_limit_monitoring agent tool and the arena determinism
driver so the live tool path and the harvested path are byte-identical.
"""
from __future__ import annotations

from typing import Any

from sqlalchemy import select
from sqlalchemy.orm import Session

from ...models import RiskRun
from .errors import LimitValidationError

_TERMINAL = ("completed", "completed_with_errors")


def derive_monitoring_envelope(session: Session, portfolio_id: int) -> dict[str, Any]:
    run = session.execute(
        select(RiskRun)
        .where(RiskRun.portfolio_id == portfolio_id, RiskRun.status.in_(_TERMINAL))
        .order_by(RiskRun.id.desc())
        .limit(1)
    ).scalar_one_or_none()
    if run is None:
        raise LimitValidationError(
            f"no completed risk run for portfolio {portfolio_id} — run risk first"
        )
    metadata = _metadata(run)
    evidence_id = metadata.get("effective_market_evidence_id")
    if not isinstance(evidence_id, str) or not evidence_id:
        raise LimitValidationError(
            f"latest risk run {run.id} has no market-evidence id — re-run risk first"
        )
    return {
        "pricing_parameter_profile_id": run.pricing_parameter_profile_id,
        "engine_config_id": run.engine_config_id,
        "market_snapshot_id": None,
        "effective_market_evidence_id": evidence_id,
        "valuation_as_of": _source_valuation(run),
        "max_source_age_seconds": None,
    }
```

Implement `_metadata` / `_source_valuation` by importing the existing helpers (`services/limits/source_planner.py::_source_metadata` and `source_valuation_at` — import them rather than duplicating; if `_source_metadata` is private, re-export it in source_planner rather than copying logic).

def _assert_no_foreign_active_limits(session, bundle) -> None:
    """Fail match setup if a foreign active limit would join this run.

    monitoring._active_versions pulls every active non-portfolio-scoped
    version into ANY portfolio's run, so a foreign global limit silently
    contaminates evaluations/incidents. Convert that drift into an
    explicit setup failure (same family as _assert_trap_sets_absent).
    """
    seeded_keys = {r["key"] for r in bundle.seed.get("risk_limits", []) if r.get("key")}
    if not seeded_keys:
        return
    from ...models import RiskLimit, RiskLimitVersion
    rows = session.execute(
        select(RiskLimit.key)
        .join(RiskLimitVersion, RiskLimitVersion.risk_limit_id == RiskLimit.id)
        .where(
            RiskLimitVersion.activated_at.is_not(None),
            RiskLimitVersion.scope_type != "portfolio",
            RiskLimit.key.not_in(seeded_keys),
        )
    ).scalars().all()
    if rows:
        raise RuntimeError(
            "arena setup: foreign active non-portfolio limit versions would join "
            f"this match's monitoring run: {sorted(set(rows))} — retire or "
            "deactivate them before running this workflow"
        )
